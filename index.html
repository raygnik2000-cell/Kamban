// URL de la Web App desplegada en Apps Script const GAS_API_URL = "https://script.google.com/macros/s/AKfycbznzgTWD5tbwRLQ9uMv_Dc4qU1QGZcx23fzNVEm7kBUpFjFJfb1eSk1wbrfzaOjdQr3/exec; // <--- Reemplaza con tu ejecutable let tasks = []; document.addEventListener("DOMContentLoaded", () => { fetchTasks(); setupDragAndDrop(); document.getElementById("task-form").addEventListener("submit", handleCreateTask); }); async function fetchTasks() { try { const res = await fetch(GAS_API_URL); tasks = await res.json(); renderBoard(); } catch (err) { console.error("Error al cargar tareas:", err); } } function renderBoard() { const columns = document.querySelectorAll(".kanban-col"); columns.forEach(col => { const status = col.dataset.status; const cardList = col.querySelector(".card-list"); const countSpan = col.querySelector(".count"); cardList.innerHTML = ""; const filteredTasks = tasks.filter(t => t.status === status); countSpan.textContent = filteredTasks.length; filteredTasks.forEach(task => { cardList.appendChild(createCardElement(task)); }); }); } function createCardElement(task) { const card = document.createElement("div"); card.className = "bg-white p-4 rounded-lg shadow-sm border border-slate-200 cursor-grab hover:shadow-md transition space-y-2"; card.draggable = true; card.dataset.id = task.id; card.innerHTML = `

${task.title}

${task.description || 'Sin descripción'}

${task.assignee} ${task.endDate || 'S/F'}
${task.url ? `🔗 Entregable` : ''} `; card.addEventListener("dragstart", (e) => { card.classList.add("dragging"); e.dataTransfer.setData("text/plain", task.id); }); card.addEventListener("dragend", () => { card.classList.remove("dragging"); }); return card; } function setupDragAndDrop() { const columns = document.querySelectorAll(".kanban-col"); columns.forEach(col => { col.addEventListener("dragover", (e) => e.preventDefault()); col.addEventListener("drop", async (e) => { e.preventDefault(); const taskId = e.dataTransfer.getData("text/plain"); const newStatus = col.dataset.status; const task = tasks.find(t => t.id.toString() === taskId); if (task && task.status !== newStatus) { task.status = newStatus; renderBoard(); // Actualización asíncrona a Google Sheets await fetch(GAS_API_URL, { method: "POST", mode: "no-cors", // Requerido por Apps Script al hacer POST directo headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "updateStatus", id: taskId, newStatus }) }); } }); }); } async function handleCreateTask(e) { e.preventDefault(); const newTask = { id: Date.now(), title: document.getElementById("title").value, description: document.getElementById("description").value, assignee: document.getElementById("assignee").value, status: document.getElementById("status").value, startDate: document.getElementById("startDate").value, endDate: document.getElementById("endDate").value, url: document.getElementById("url").value }; tasks.push(newTask); renderBoard(); toggleModal(false); e.target.reset(); await fetch(GAS_API_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "createTask", task: newTask }) }); } function toggleModal(show) { document.getElementById("modal").classList.toggle("hidden", !show); }
